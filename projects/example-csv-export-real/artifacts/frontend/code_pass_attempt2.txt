  isOpen: boolean;
  onClose: () => void;
  selectedMonth: string; // Format: YYYY-MM
}

  isOpen, 
  onClose, 
  selectedMonth 
}) => {
  const { status, error, isDownloading, handleExport, reset } = useExport();
  
  // Reset state when modal opens/closes to avoid stale UI
  useEffect(() => {
    if (!isOpen) {
      reset();
    }
  }, [isOpen, reset]);

  if (!isOpen) return null;

  const handleConfirm = async () => {
    await handleExport(selectedMonth);
  };

  const handleClose = () => {
    reset();
    onClose();
  };

  let feedbackMessage = '';
  let feedbackType: 'info' | 'success' | 'error' = 'info';

    feedbackMessage = 'Export started. Check your downloads folder.';

    feedbackMessage = error.message;
