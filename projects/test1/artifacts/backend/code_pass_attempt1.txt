"""
Dungeon Generator Service
Deterministic dungeon generation based on seed, ensuring connectivity and room types.
"""

import random
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass, field

# --- Constants & Config ---

ROOM_TYPES = {
    "START": "start",
    "COMBAT": "combat",
    "PLATFORMING": "platforming",
    "TREASURE": "treasure",
    "SHOP": "shop",
    "BONFIRE": "bonfire",
    "EVENT": "event",
    "ELITE": "elite",
    "BOSS": "boss"
}

# Chapter 1 Configuration (MVP Scope)
CHAPTER_CONFIG = {
    "name": "蘑菇林迷宫",
    "total_layers": 5,
    "rooms_per_layer": 8,  # MVP simplified to 8 rooms + 1 boss
    "elite_probability": 0.6,
    "boss_room_index": -1  # Always last
}

# Room Template Definitions (Simplified for Backend Logic)
ROOM_TEMPLATES = {
    "combat": {"min_waves": 2, "max_waves": 4},
    "platforming": {"has_chest": True, "trap_chance": 0.3},
    "treasure": {"min_chests": 1, "max_chests": 3},
    "shop": {"sells_equipment": True, "sells_keys": True},
    "bonfire": {"heals_player": True, "allows_save": True},
    "event": {"type": "gamble_or_curse"},
    "elite": {"has_affixes": True, "guaranteed_blue_plus_drop": True},
    "boss": {"phase_count": 3, "guaranteed_blue_plus_drop": True}
}

@dataclass
class Room:
    room_id: str
    room_type: str
    layer_index: int
    position: Tuple[int, int]  # Grid coordinates (x, y)
    connections: List[str] = field(default_factory=list)
    data: Dict = field(default_factory=dict)

@dataclass
class DungeonLayer:
    layer_index: int
    rooms: List[Room]
    start_room_id: str
    boss_room_id: str
    seed: int

@dataclass
class GeneratedDungeon:
    chapter_name: str
    layers: List[DungeonLayer]
    global_seed: int

class DungeonGenerator:
    """
    Generates deterministic dungeons based on a seed.
    Ensures connectivity from Start to Boss via a main path, with branches for side rooms.
    """

    def __init__(self):
        self._rng = random.Random()

    def generate_dungeon(self, chapter_id: str, global_seed: int) -> GeneratedDungeon:
        """
        Main entry point for dungeon generation.
        """
        config = CHAPTER_CONFIG  # In production, fetch from ConfigService
        
        rng = random.Random(global_seed)
        layers = []

        for layer_idx in range(config["total_layers"]):
            layer_rooms = self._generate_layer(layer_idx, config, rng)
            layers.append(DungeonLayer(
                layer_index=layer_idx,
                rooms=layer_rooms,
                start_room_id=layer_rooms[0].room_id,
                boss_room_id=layer_rooms[-1].room_id,
                seed=rng.randint(0, 2**32)
            ))

        return GeneratedDungeon(
            chapter_name=config["name"],
            layers=layers,
            global_seed=global_seed
        )

    def _generate_layer(self, layer_idx: int, config: dict, rng: random.Random) -> List[Room]:
        """
        Generates a single layer of rooms.
        Structure: Linear path for main progression, with side branches.
        """
        num_rooms = config["rooms_per_layer"]
        rooms = []
        
        # 1. Define Main Path Indices (0 to num_rooms-2)
        # The last room is always the Boss
        main_path_indices = list(range(num_rooms - 1))
        
        # 2. Assign Room Types along Main Path
        # Ensure at least one Bonfire and one Shop in the layer if possible
        has_bonfire = False
        has_shop = False
        
        for i in range(len(main_path_indices)):
            room_idx = main_path_indices[i]
            
            # Determine type based on position and probability
            if i == 0:
                r_type = ROOM_TYPES["START"]
            elif i == len(main_path_indices) - 1:
                # Second to last is usually Elite or Combat leading to Boss
                if rng.random() < config.get("elite_probability", 0.6):
                    r_type = ROOM_TYPES["ELITE"]
                else:
                    r_type = ROOM_TYPES["COMBAT"]
            else:
                # Random assignment for intermediate rooms
                roll = rng.random()
                if not has_bonfire and i == len(main_path_indices) // 2:
                    r_type = ROOM_TYPES["BONFIRE"]
                    has_bonfire = True
                elif not has_shop and i == (len(main_path_indices) // 4):
                    r_type = ROOM_TYPES["SHOP"]
                    has_shop = True
                else:
                    # Weighted random for other types
                    if roll < 0.4:
                        r_type = ROOM_TYPES["COMBAT"]
                    elif roll < 0.6:
                        r_type = ROOM_TYPES["PLATFORMING"]
                    elif roll < 0.8:
                        r_type = ROOM_TYPES["TREASURE"]
                    else:
                        r_type = ROOM_TYPES["EVENT"]

            # Create Room Object
            room_id = f"L{layer_idx}_R{room_idx}"
            position = (i, 0) # Linear layout for MVP
            
            # Generate specific data based on type
            data = self._generate_room_data(r_type, rng)
            
            rooms.append(Room(
                room_id=room_id,
                room_type=r_type,
                layer_index=layer_idx,
                position=position,
                connections=[],
                data=data
            ))

        # 3. Add Boss Room (Last one)
        boss_room = Room(
            room_id=f"L{layer_idx}_R{num_rooms-1}",
            room_type=ROOM_TYPES["BOSS"],
            layer_index=layer_idx,
            position=(num_rooms - 1, 0),
            connections=[],
            data={"phase_count": 3}
        )
        rooms.append(boss_room)

        # 4. Establish Connections (Linear Chain)
        for i in range(len(rooms) - 1):
            rooms[i].connections.append(rooms[i+1].room_id)
            rooms[i+1].connections.append(rooms[i].room_id)

        return rooms

    def _generate_room_data(self, r_type: str, rng: random.Random) -> Dict:
        """Generates specific data payload for a room type."""
        if r_type == ROOM_TYPES["COMBAT"]:
            waves = rng.randint(2, 4)
            return {"waves": [{"count": rng.randint(3, 6), "type": "slime"} for _ in range(waves)]}
        
        elif r_type == ROOM_TYPES["TREASURE"]:
            chests = rng.randint(1, 3)
            return {"chests": [self._generate_chest(rng) for _ in range(chests)]}
        
        elif r_type == ROOM_TYPES["EVENT"]:
            event_type = rng.choice(["gamble", "curse_blessing"])
            return {"event_type": event_type, "resolved": False}
        
        elif r_type == ROOM_TYPES["ELITE"]:
            affixes = ["shield", "speed"] if rng.random() > 0.5 else ["regen"]
            return {"affixes": affixes, "elite_hp_mult": 2.0}
        
        return {}

    def _generate_chest(self, rng: random.Random) -> Dict:
        """Helper to generate chest data."""
        is_trap = rng.random() < 0.3
        return {
            "id": f"chest_{rng.randint(1000, 9999)}",
            "is_trap": is_trap,
            "loot_table_id": "common_treasure" if not is_trap else None
        }

    def get_room_by_id(self, dungeon: GeneratedDungeon, room_id: str) -> Optional[Room]:
        """Helper to find a specific room in the generated dungeon."""
        for layer in dungeon.layers:
            for room in layer.rooms:
                if room.room_id == room_id:
                    return room
        return None

    def get_next_rooms(self, dungeon: GeneratedDungeon, current_room_id: str) -> List[Room]:
        """Get valid next rooms from a specific room."""
        current = self.get_room_by_id(dungeon, current_room_id)
        if not current:
            return []
        
        next_rooms = []
        for conn_id in current.connections:
            # Filter out locked doors or completed boss rooms if necessary
            # For MVP, just return connected rooms
            room = self.get_room_by_id(dungeon, conn_id)
            if room and not self._is_room_locked(room):
                next_rooms.append(room)
        return next_rooms

    def _is_room_locked(self, room: Room) -> bool:
        """Check if a room is locked (e.g., requires key)."""
        # MVP: Only keys might lock treasure rooms in some designs, 
        # but spec says "max 1 key door per layer". 
        # For now, assume unlocked unless data indicates otherwise.
        return False
